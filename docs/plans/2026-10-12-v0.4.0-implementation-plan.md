# v0.4.0 Sign in Through the Browser: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The widget reads usage with an Anthropic OAuth token, obtained either from Claude Code's Keychain item or from a sign-in in the user's default browser, with the v0.3.0 embedded login kept as a hidden fallback.

**Architecture:** `OAuthClient` (URLSession) talks to the authorize, token, usage and profile endpoints. `CallbackServer` (NWListener on 127.0.0.1) catches the browser redirect once. `AuthManager` becomes a coordinator that resolves the identity in order (own token, Claude Code's token, web session) and runs the browser flow; the v0.3.0 state machine is renamed `WebSessionAuth` and used as the fallback. `UsageService` polls a `UsageSource`, of which there are two.

**Tech Stack:** Swift 5.9 language mode, AppKit, Network framework (`NWListener`), CryptoKit (SHA-256), Security framework (Keychain), URLSession, XCTest with a stub `URLProtocol`, xcodegen 2.46.

**Spec:** `docs/plans/2026-10-12-v0.4.0-browser-sign-in-design.md`

## Global Constraints

- Deployment target stays `macOS 13.0`.
- `project.yml` is the source of truth; after editing it run `xcodegen generate` and commit the regenerated `.xcodeproj`. New files also need `xcodegen generate` before they compile.

- Every OAuth request carries `anthropic-beta: oauth-2025-04-20`, `User-Agent: ClaudeUsageWidget/<version> (macOS)`, `Accept: application/json`. Never any other User-Agent.

- Tokens live in the Keychain or in memory, never in UserDefaults, logs or files.
- Tests never touch the real Keychain items `io.rsla.ClaudeUsageWidget` or `Claude Code-credentials`; the one Keychain test uses service `io.rsla.ClaudeUsageWidget.tests` and deletes it.

```swift
import CryptoKit
import Foundation
import Security

/// RFC 7636 code verifier and S256 challenge, plus the random `state` for the authorize request.
struct PKCE: Equatable {
    let verifier: String
    let challenge: String

    static func generate() -> PKCE {
        let verifier = randomBytes(32).base64URLEncoded()
        return PKCE(verifier: verifier, challenge: challenge(for: verifier))
    }

    static func challenge(for verifier: String) -> String {
        Data(SHA256.hash(data: Data(verifier.utf8))).base64URLEncoded()
    }

    static func randomState() -> String {
        randomBytes(16).base64URLEncoded()
    }

    private static func randomBytes(_ count: Int) -> Data {
        var bytes = [UInt8](repeating: 0, count: count)
        let status = SecRandomCopyBytes(kSecRandomDefault, count, &bytes)
        precondition(status == errSecSuccess, "SecRandomCopyBytes failed: \(status)")
        return Data(bytes)
    }
}

extension Data {
    /// Base64 without padding, with the URL-safe alphabet (RFC 4648 section 5).
    func base64URLEncoded() -> String {
        base64EncodedString()
            .replacingOccurrences(of: "+", with: "-")
            .replacingOccurrences(of: "/", with: "_")
            .replacingOccurrences(of: "=", with: "")
    }
}
```

```swift
import Foundation

/// Answers every request made through a session configured with it. Records requests for assertions.
final class StubURLProtocol: URLProtocol {
    nonisolated(unsafe) static var handler: ((URLRequest) -> (status: Int, body: Data))?
    nonisolated(unsafe) static var requests: [URLRequest] = []

    static func reset() { handler = nil; requests = [] }

    static func session() -> URLSession {
        let config = URLSessionConfiguration.ephemeral
        config.protocolClasses = [StubURLProtocol.self]
        return URLSession(configuration: config)
    }

    override class func canInit(with request: URLRequest) -> Bool { true }
    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }

    override func startLoading() {
        Self.requests.append(request)
        let answer = Self.handler?(request) ?? (500, Data())
        let response = HTTPURLResponse(url: request.url!, statusCode: answer.status, httpVersion: "HTTP/1.1", headerFields: ["Content-Type": "application/json"])!
        client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
        client?.urlProtocol(self, didLoad: answer.body)
        client?.urlProtocolDidFinishLoading(self)
    }

    override func stopLoading() {}
}

extension URLRequest {
    /// URLSession hands URLProtocol the body as a stream; read it back for assertions.
    var bodyData: Data? {
        if let httpBody { return httpBody }
        guard let stream = httpBodyStream else { return nil }
        stream.open()
        defer { stream.close() }
        var data = Data()
        var buffer = [UInt8](repeating: 0, count: 4096)
        while stream.hasBytesAvailable {
            let read = stream.read(&buffer, maxLength: buffer.count)
            if read <= 0 { break }
            data.append(buffer, count: read)
        }
        return data
    }
}
```

```swift
import Foundation

// MARK: - Configuration

/// Anthropic's consumer OAuth, as used by Claude Code (client id read from Claude Code 2.1.296; no third-party registration exists).
enum OAuthConfig {
    static let clientId = "9d1c250a-e61b-44d9-88ed-5944d1962f5e"
    static let authorizeURL = URL(string: "https://claude.com/cai/oauth/authorize")!
    static let tokenURL = URL(string: "https://platform.claude.com/v1/oauth/token")!
    static let usageURL = URL(string: "https://api.anthropic.com/api/oauth/usage")!
    static let profileURL = URL(string: "https://api.anthropic.com/api/oauth/profile")!
    static let scopes = "user:profile user:inference"
    static let betaHeader = "oauth-2025-04-20"
    static var userAgent: String { "ClaudeUsageWidget/\(AppInfo.version) (macOS)" }
}

// MARK: - TokenSet

struct TokenSet: Codable, Equatable {
    let accessToken: String
    let refreshToken: String
    let expiresAt: Date
    let scope: String?

    var secondsUntilExpiry: TimeInterval { expiresAt.timeIntervalSinceNow }

    private struct Raw: Decodable {
        let access_token: String
        let refresh_token: String
        let expires_in: Double
        let scope: String?
    }

    static func decode(from data: Data, now: Date) throws -> TokenSet {
        let raw = try JSONDecoder().decode(Raw.self, from: data)
        return TokenSet(accessToken: raw.access_token, refreshToken: raw.refresh_token, expiresAt: now.addingTimeInterval(raw.expires_in), scope: raw.scope)
    }

    /// `{"error":"invalid_grant","error_description":"..."}` or the API envelope; falls back to a generic message.
    static func refusalMessage(in data: Data) -> String {
        guard let object = try? JSONSerialization.jsonObject(with: data) as? [String: Any] else { return "token request refused" }
        if let description = object["error_description"] as? String { return description }
        if let error = object["error"] as? String { return error }
        if let error = object["error"] as? [String: Any], let message = error["message"] as? String { return message }
        return "token request refused"
    }
}

// MARK: - Profile

struct Profile: Codable, Equatable {
    let email: String?
    let displayName: String?
    let organizationName: String?
    let organizationType: String?

    private struct Raw: Decodable {
        struct Account: Decodable { let email: String?; let display_name: String? }
        struct Organization: Decodable { let name: String?; let organization_type: String? }
        let account: Account?
        let organization: Organization?
    }

    static func decode(from data: Data) throws -> Profile {
        let raw = try JSONDecoder().decode(Raw.self, from: data)
        return Profile(email: raw.account?.email, displayName: raw.account?.display_name, organizationName: raw.organization?.name, organizationType: raw.organization?.organization_type)
    }
}

// MARK: - OAuthClient

/// Thin HTTP layer for the OAuth endpoints. Every request carries the beta header and an honest User-Agent
/// (Cloudflare in front of the token endpoint rejects anonymous-looking clients).
@MainActor
final class OAuthClient {
    private let session: URLSession
    private let now: () -> Date

    init(session: URLSession = .shared, now: @escaping () -> Date = Date.init) {
        self.session = session
        self.now = now
    }

    func authorizeURL(redirectURI: String, challenge: String, state: String) -> URL {
        var components = URLComponents(url: OAuthConfig.authorizeURL, resolvingAgainstBaseURL: false)!
        components.queryItems = [
            URLQueryItem(name: "code", value: "true"),
            URLQueryItem(name: "client_id", value: OAuthConfig.clientId),
            URLQueryItem(name: "response_type", value: "code"),
            URLQueryItem(name: "redirect_uri", value: redirectURI),
            URLQueryItem(name: "scope", value: OAuthConfig.scopes),
            URLQueryItem(name: "code_challenge", value: challenge),
```

```swift
import Foundation
import Network

@MainActor
protocol CallbackServing: AnyObject {
    /// Binds 127.0.0.1 on a free port and returns it. Only a callback carrying `expectedState` is accepted.
    func start(expectedState: String) async throws -> UInt16
    /// The authorization code, as soon as the browser delivers it (possibly before this call), or .callbackTimeout.
    func waitForCode(timeout: TimeInterval) async throws -> String
    /// Stops listening; a pending waitForCode throws .signInCancelled.
    func stop()
}

/// One-shot HTTP listener for the OAuth redirect: GET /callback?code=&state= on 127.0.0.1.
@MainActor
final class CallbackServer: CallbackServing {
    private let queue = DispatchQueue(label: "io.rsla.ClaudeUsageWidget.callback")
    private var listener: NWListener?
    private var connections: [NWConnection] = []
    private var expectedState = ""
    private var pendingCode: String?
    private var startContinuation: CheckedContinuation<UInt16, Error>?
    private var codeContinuation: CheckedContinuation<String, Error>?

    func start(expectedState: String) async throws -> UInt16 {
        self.expectedState = expectedState
        let parameters = NWParameters.tcp
        parameters.requiredLocalEndpoint = NWEndpoint.hostPort(host: "127.0.0.1", port: .any)
        let listener = try NWListener(using: parameters)
        self.listener = listener
        listener.newConnectionHandler = { [weak self] connection in
            Task { @MainActor in self?.accept(connection) }
        }
        listener.stateUpdateHandler = { [weak self] state in
            Task { @MainActor in self?.listenerStateChanged(state) }
        }
        return try await withCheckedThrowingContinuation { continuation in
            startContinuation = continuation
            listener.start(queue: queue)
        }
    }

    func waitForCode(timeout: TimeInterval) async throws -> String {
        if let code = pendingCode {
            pendingCode = nil
            return code
        }
        let timeoutTask = Task { [weak self] in
            try? await Task.sleep(nanoseconds: UInt64(timeout * 1_000_000_000))
            guard !Task.isCancelled else { return }
            await self?.finishWaiting(with: .failure(ClaudeAPIError.callbackTimeout))
        }
        defer { timeoutTask.cancel() }
        return try await withCheckedThrowingContinuation { continuation in
            codeContinuation = continuation
        }
    }

    func stop() {
        finishWaiting(with: .failure(ClaudeAPIError.signInCancelled))
        pendingCode = nil
        listener?.cancel()
        listener = nil
        connections.forEach { $0.cancel() }
        connections.removeAll()
    }

    // MARK: - Listener

    private func listenerStateChanged(_ state: NWListener.State) {
        switch state {
        case .ready:
            startContinuation?.resume(returning: listener?.port?.rawValue ?? 0)
            startContinuation = nil
        case .failed(let error):
            startContinuation?.resume(throwing: ClaudeAPIError.transport("Callback listener failed: \(error.localizedDescription)"))
            startContinuation = nil
            finishWaiting(with: .failure(ClaudeAPIError.transport("Callback listener failed")))
        default:
            break
        }
    }

    private func finishWaiting(with result: Result<String, Error>) {
        guard let continuation = codeContinuation else { return }
        codeContinuation = nil
        continuation.resume(with: result)
    }

    // MARK: - Connections

    private func accept(_ connection: NWConnection) {
        connections.append(connection)
        connection.start(queue: queue)
        read(connection, buffer: Data())
    }

    private func read(_ connection: NWConnection, buffer: Data) {
        connection.receive(minimumIncompleteLength: 1, maximumLength: 16_384) { [weak self] data, _, isComplete, error in
            Task { @MainActor in
                guard let self else { return }
                var buffer = buffer
                if let data { buffer.append(data) }
                if let terminator = buffer.range(of: Data("\r\n\r\n".utf8)) {
                    self.handle(head: String(decoding: buffer[..<terminator.lowerBound], as: UTF8.self), on: connection)
                } else if error != nil || isComplete || buffer.count >= 16_384 {
                    connection.cancel()
                } else {
                    self.read(connection, buffer: buffer)
                }
            }
        }
    }

    private func handle(head: String, on connection: NWConnection) {
        let requestLine = head.split(separator: "\r\n", maxSplits: 1).first.map(String.init) ?? ""
        let parts = requestLine.split(separator: " ")
        let target = parts.count >= 2 ? String(parts[1]) : "/"
        let components = URLComponents(string: "http://localhost\(target)")
        let path = components?.path ?? ""
        let query = components?.queryItems ?? []
        let code = query.first { $0.name == "code" }?.value ?? ""
        let state = query.first { $0.name == "state" }?.value

        guard path == "/callback" else {
            respond(connection, status: "404 Not Found", html: "<p>Not found.</p>")
            return
        }
        guard !code.isEmpty, state == expectedState else {
            respond(connection, status: "400 Bad Request", html: "<h2>Sign-in response did not match.</h2><p>Go back to the widget and try again.</p>")
            return
        }
        respond(connection, status: "200 OK", html: "<h2>Signed in to Claude Usage Widget.</h2><p>You can close this tab.</p>")
        if codeContinuation != nil {
            finishWaiting(with: .success(code))
        } else {
            pendingCode = code
        }
    }

    private func respond(_ connection: NWConnection, status: String, html: String) {
        let body = "<!doctype html><html><head><meta charset=\"utf-8\"><title>Claude Usage Widget</title></head><body style=\"font-family:-apple-system,sans-serif;padding:40px\">\(html)</body></html>"
        let head = "HTTP/1.1 \(status)\r\nContent-Type: text/html; charset=utf-8\r\nContent-Length: \(body.utf8.count)\r\nConnection: close\r\n\r\n"
        connection.send(content: Data((head + body).utf8), completion: .contentProcessed { _ in connection.cancel() })
    }
}
```

```swift
import Foundation
import Security

/// Where the widget's own OAuth tokens live. Implemented by the Keychain in the app and in memory in tests.
protocol TokenStoring: AnyObject {
    func load() throws -> TokenSet?
    func save(_ tokens: TokenSet) throws
    func clear() throws
}

final class KeychainTokenStore: TokenStoring {
    private let service: String
    private let account: String

    init(service: String = "io.rsla.ClaudeUsageWidget", account: String = "oauth") {
        self.service = service
        self.account = account
    }

    func load() throws -> TokenSet? {
        guard let data = try KeychainItem.read(service: service, account: account) else { return nil }
        do {
            return try JSONDecoder().decode(TokenSet.self, from: data)
        } catch {
            try? clear()   // unreadable item: drop it, the user signs in again
            return nil
        }
    }

    func save(_ tokens: TokenSet) throws {
        try KeychainItem.write(try JSONEncoder().encode(tokens), service: service, account: account)
    }

    func clear() throws {
        try KeychainItem.delete(service: service, account: account)
    }
}

/// Generic-password helpers. A missing item reads as nil; any other failure is ClaudeAPIError.keychain(status).
enum KeychainItem {
    static func read(service: String, account: String?) throws -> Data? {
        var query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]
        if let account { query[kSecAttrAccount as String] = account }
        var item: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &item)
        if status == errSecItemNotFound { return nil }
        guard status == errSecSuccess else { throw ClaudeAPIError.keychain(status) }
        return item as? Data
    }

    static func write(_ data: Data, service: String, account: String) throws {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: account,
        ]
        let status = SecItemUpdate(query as CFDictionary, [kSecValueData as String: data] as CFDictionary)
        if status == errSecItemNotFound {
            var add = query
            add[kSecValueData as String] = data
            let addStatus = SecItemAdd(add as CFDictionary, nil)
            guard addStatus == errSecSuccess else { throw ClaudeAPIError.keychain(addStatus) }
        } else if status != errSecSuccess {
            throw ClaudeAPIError.keychain(status)
        }
    }

    static func delete(service: String, account: String) throws {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: account,
        ]
        let status = SecItemDelete(query as CFDictionary)
        guard status == errSecSuccess || status == errSecItemNotFound else { throw ClaudeAPIError.keychain(status) }
    }
}
```

```swift
    let expiresAt: Date
    let subscriptionType: String?

    /// Usable for at least another minute.
    var isValid: Bool { expiresAt.timeIntervalSinceNow > 60 }

    private struct Raw: Decodable {
        struct OAuth: Decodable {
            let accessToken: String
            let expiresAt: Double        // milliseconds since the epoch (JavaScript Date.now())
            let subscriptionType: String?
        }
        let claudeAiOauth: OAuth?
    }

    static func parse(_ data: Data) throws -> ClaudeCodeToken? {
        let raw = try JSONDecoder().decode(Raw.self, from: data)

        return ClaudeCodeToken(accessToken: oauth.accessToken, expiresAt: Date(timeIntervalSince1970: oauth.expiresAt / 1000), subscriptionType: oauth.subscriptionType)
    }
}
```

```swift
    case ownToken      // the widget's own OAuth token (browser sign-in)
    case claudeCode    // Claude Code's token, read from its Keychain item
    case webSession    // v0.3.0 claude.ai cookies in the hidden WKWebView

/// Something that can fetch the raw usage response for the current identity.
@MainActor
protocol UsageSource: AnyObject {
    var identity: IdentitySource { get }
    func fetchUsage() async throws -> APIResponse
}

/// Hands out a bearer token, refreshing when it can. Throws .sessionExpired when there is nothing usable.
@MainActor
protocol BearerTokenProvider: AnyObject {
    func accessToken() async throws -> String
    /// After the API answered 401: a fresh token, or .sessionExpired.
    func refreshAfterUnauthorized() async throws -> String
}

// MARK: - Own token

@MainActor
final class OwnTokenProvider: BearerTokenProvider {
    private let store: TokenStoring
    private let client: OAuthClient
    private let refreshLeeway: TimeInterval = 300
    private var refreshTask: Task<TokenSet, Error>?

    init(store: TokenStoring, client: OAuthClient) {
        self.store = store
        self.client = client
    }

    func accessToken() async throws -> String {
        guard let tokens = try store.load() else { throw ClaudeAPIError.sessionExpired }
        if tokens.secondsUntilExpiry > refreshLeeway { return tokens.accessToken }
        return try await refresh(tokens).accessToken
    }

    func refreshAfterUnauthorized() async throws -> String {
        guard let tokens = try store.load() else { throw ClaudeAPIError.sessionExpired }
        return try await refresh(tokens).accessToken
    }

    /// One refresh in flight at a time; concurrent callers share it. A refused refresh clears the store.
    private func refresh(_ tokens: TokenSet) async throws -> TokenSet {
        if let task = refreshTask { return try await task.value }
        let task = Task<TokenSet, Error> { [client, store] in
            do {
                let fresh = try await client.refresh(tokens.refreshToken)
                try store.save(fresh)
                return fresh
            } catch ClaudeAPIError.oauth {
                try? store.clear()
                throw ClaudeAPIError.sessionExpired
            }
        }
        refreshTask = task
        defer { refreshTask = nil }
        return try await task.value
    }
}

// MARK: - Claude Code's token

@MainActor
final class ClaudeCodeTokenProvider: BearerTokenProvider {
    private let reader: ClaudeCodeCredentialsReading

    init(reader: ClaudeCodeCredentialsReading) {
        self.reader = reader
    }

    func accessToken() async throws -> String {
        guard let token = try reader.read(), token.isValid else { throw ClaudeAPIError.sessionExpired }
        return token.accessToken
    }

    /// Claude Code may have refreshed since the last read; read again, never refresh it ourselves.
    func refreshAfterUnauthorized() async throws -> String {
        try await accessToken()
    }
}
```

```swift
import AppKit
import Foundation

enum AuthState: Equatable {
    case unknown
    case unreachable
    case signedOut(claudeCodeAvailable: Bool)     // which sign-in buttons to show
    case signingIn                                 // browser flow waiting for the redirect
    case signedIn(IdentitySource, Profile?)
    case sessionExpired(IdentitySource)
    case noChatOrg                                 // web session only
}

@MainActor
protocol AuthManagerDelegate: AnyObject {
    func authManager(_ manager: AuthManager, didChange state: AuthState)
}

/// Picks the identity the widget runs on (own OAuth token, Claude Code's token, or the v0.3.0 web session),
/// runs the sign-in flows, and owns the usage source for the current identity.
@MainActor
final class AuthManager {

    struct Dependencies {
        var oauth: OAuthClient
        var tokenStore: TokenStoring
        var claudeCode: ClaudeCodeCredentialsReading
        var webSession: WebSessionAuth
        var defaults: UserDefaults
        var makeCallbackServer: @MainActor () -> CallbackServing
        var openURL: (URL) -> Void
        var hasWebSessionCookies: () async -> Bool
    }

    static let optOutKey = "claudeCodeTokenOptOut"
    static let deniedKey = "claudeCodeTokenDenied"
    static let webSessionKey = "webSessionEnabled"

    weak var delegate: AuthManagerDelegate?

    private(set) var state: AuthState = .unknown {
        didSet {
            guard oldValue != state else { return }
            delegate?.authManager(self, didChange: state)
            if state == .unreachable { scheduleRetry() } else { cancelRetry() }
        }
    }

    /// The source to poll while `.signedIn`; nil otherwise.
    private(set) var usageSource: UsageSource?
    /// Why the last sign-in attempt ended, for the popover. Cleared on the next attempt.
    private(set) var signInMessage: String?

    var unreachableRetryInterval: TimeInterval = 60
    var browserTimeout: TimeInterval = 300
    var isRetryScheduled: Bool { retryTimer != nil }

    private let deps: Dependencies
    private var claudeCodeItemExists = false
    private var browserFlow: CallbackServing?
    private var retryTimer: Timer?

    init(dependencies: Dependencies) {
        deps = dependencies
        deps.webSession.delegate = self
    }

    // MARK: - Resolution

    func checkAuthStatus() async {
        if await resolveOwnToken() { return }
        if await resolveClaudeCode() { return }
        if await resolveWebSession() { return }
        usageSource = nil
        state = .signedOut(claudeCodeAvailable: claudeCodeItemExists)
    }

    /// True when the own token produced a terminal state (signed in, or unreachable while refreshing it).
    private func resolveOwnToken() async -> Bool {
        let provider = OwnTokenProvider(store: deps.tokenStore, client: deps.oauth)
        do {
            let token = try await provider.accessToken()
            let profile = try? await deps.oauth.profile(accessToken: token)
            usageSource = OAuthUsageSource(client: deps.oauth, tokens: provider, identity: .ownToken)
            state = .signedIn(.ownToken, profile)
            return true
        } catch ClaudeAPIError.sessionExpired {
            return false            // no own token, or the refresh was refused (store already cleared)
        } catch ClaudeAPIError.transport {
            usageSource = nil
            state = .unreachable
            return true
        } catch {
            return false            // unreadable store: fall through to the next source
        }
    }

    private func resolveClaudeCode() async -> Bool {
        guard !deps.defaults.bool(forKey: Self.optOutKey), !deps.defaults.bool(forKey: Self.deniedKey) else { return false }
        do {
            guard let token = try deps.claudeCode.read() else {
                claudeCodeItemExists = false
                return false
            }
            claudeCodeItemExists = true
            guard token.isValid else { return false }
            let profile = try? await deps.oauth.profile(accessToken: token.accessToken)
            usageSource = OAuthUsageSource(client: deps.oauth, tokens: ClaudeCodeTokenProvider(reader: deps.claudeCode), identity: .claudeCode)
            state = .signedIn(.claudeCode, profile)
            return true
        } catch {
            // The item exists and macOS refused us (Deny). Remember it so the prompt does not return every poll.
            claudeCodeItemExists = true
            deps.defaults.set(true, forKey: Self.deniedKey)
            return false
        }
    }

    private func resolveWebSession() async -> Bool {
        if !deps.defaults.bool(forKey: Self.webSessionKey), await deps.hasWebSessionCookies() {
            deps.defaults.set(true, forKey: Self.webSessionKey)   // upgrading from v0.3.0 keeps the user signed in
        }
        guard deps.defaults.bool(forKey: Self.webSessionKey) else { return false }
        return apply(webState: await deps.webSession.verifySession())
    }

    /// Maps the web-session machine onto ours. False only when it is simply signed out.
    @discardableResult
    private func apply(webState: WebSessionState) -> Bool {
        switch webState {
        case .signedIn(let org, let chatOrgCount):
            usageSource = WebSessionUsageSource(client: deps.webSession.client, orgId: org.uuid)
            let profile = chatOrgCount > 1 ? Profile(email: nil, displayName: nil, organizationName: org.displayName, organizationType: nil) : nil
            state = .signedIn(.webSession, profile)
            return true
        case .noChatOrg:
            usageSource = nil
            state = .noChatOrg
            return true
        case .sessionExpired:
            usageSource = nil
            state = .sessionExpired(.webSession)
            return true
        case .unreachable:
            usageSource = nil
            state = .unreachable
            return true
        case .signedOut, .unknown:
            return false
        }
    }

    // MARK: - Browser sign-in

    func signInWithBrowser() async {
        cancelBrowserSignIn()
        signInMessage = nil
        let pkce = PKCE.generate()
        let stateToken = PKCE.randomState()
        let server = deps.makeCallbackServer()
        browserFlow = server
        state = .signingIn
        do {
            let port = try await server.start(expectedState: stateToken)
            let redirectURI = "http://localhost:\(port)/callback"
            deps.openURL(deps.oauth.authorizeURL(redirectURI: redirectURI, challenge: pkce.challenge, state: stateToken))
            let code = try await server.waitForCode(timeout: browserTimeout)
            server.stop()
            browserFlow = nil
            let tokens = try await deps.oauth.exchange(code: code, verifier: pkce.verifier, redirectURI: redirectURI, state: stateToken)
            try deps.tokenStore.save(tokens)
            deps.defaults.set(false, forKey: Self.optOutKey)
            await checkAuthStatus()
        } catch {
            server.stop()
            browserFlow = nil
            signInMessage = ((error as? ClaudeAPIError) ?? .transport(error.localizedDescription)).userMessage
            usageSource = nil
            state = .signedOut(claudeCodeAvailable: claudeCodeItemExists)
        }
    }

    func cancelBrowserSignIn() {
        browserFlow?.stop()
        browserFlow = nil
    }

    // MARK: - Claude Code

    func useClaudeCodeToken() async {
        deps.defaults.set(false, forKey: Self.optOutKey)
        deps.defaults.set(false, forKey: Self.deniedKey)
        await checkAuthStatus()
    }

    // MARK: - Embedded login (fallback)

    func signInInsideApp() {
        deps.defaults.set(true, forKey: Self.webSessionKey)
        deps.webSession.showLoginWindow()
    }

    // MARK: - Expiry

    /// From UsageService: the current identity no longer works.
    func sessionDidExpire() {
        guard case .signedIn(let source, _) = state else { return }
        usageSource = nil
        switch source {
        case .ownToken:
            try? deps.tokenStore.clear()
            state = .sessionExpired(.ownToken)
        case .claudeCode:
            state = .sessionExpired(.claudeCode)
        case .webSession:
            deps.webSession.sessionDidExpire()   // opens the embedded login once; its delegate maps the state
        }
    }

    // MARK: - Sign out

    func signOut() {
        cancelBrowserSignIn()
        try? deps.tokenStore.clear()
        deps.defaults.set(true, forKey: Self.optOutKey)
        if deps.defaults.bool(forKey: Self.webSessionKey) {
            deps.defaults.set(false, forKey: Self.webSessionKey)
            deps.webSession.signOut(showLoginWindow: false)
        }
        usageSource = nil
        signInMessage = nil
        state = .signedOut(claudeCodeAvailable: claudeCodeItemExists)
    }

    // MARK: - Unreachable retry

    private func scheduleRetry() {
        retryTimer?.invalidate()
        retryTimer = Timer.scheduledTimer(withTimeInterval: unreachableRetryInterval, repeats: false) { [weak self] _ in
            Task { @MainActor in
                guard let self else { return }
                self.retryTimer = nil
                await self.checkAuthStatus()
                if self.state == .unreachable { self.scheduleRetry() }
            }
        }
    }

    private func cancelRetry() {
        retryTimer?.invalidate()
        retryTimer = nil
    }
}

// MARK: - WebSessionAuthDelegate

extension AuthManager: WebSessionAuthDelegate {
    func webSessionAuth(_ auth: WebSessionAuth, didChange webState: WebSessionState) {
        guard deps.defaults.bool(forKey: Self.webSessionKey) else { return }
        if !apply(webState: webState) {
            usageSource = nil
            state = .signedOut(claudeCodeAvailable: claudeCodeItemExists)
        }
    }
}
```

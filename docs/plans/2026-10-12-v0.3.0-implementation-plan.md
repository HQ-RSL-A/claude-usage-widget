# v0.3.0 Sign-in for Anyone: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Any Claude.ai consumer account can sign in to the widget and see its own usage, with expired sessions detected and re-prompted.

**Architecture:** A WKWebView-backed `ClaudeAPIClient` returns `{status, body}` for every same-origin call and a pure `classify` maps HTTP status plus the error envelope to typed errors. `AuthManager` is a state machine proven by `/api/organizations`, picks the chat org, and owns the login window. `UsageService` polls one org and reports expiry. The popover builds rows from whatever buckets the API returned.

**Tech Stack:** Swift 5.9 language mode, AppKit, WebKit (`callAsyncJavaScript` async API), Swift concurrency on the main actor, XCTest, xcodegen 2.46, GitHub Actions `macos-15`.

**Spec:** `docs/plans/2026-10-12-v0.3.0-sign-in-for-anyone-design.md`

## Global Constraints

- Deployment target stays `macOS 13.0`; no `WKWebsiteDataStore(forIdentifier:)`.
- `project.yml` is the source of truth; after editing it run `xcodegen generate` and commit the regenerated `.xcodeproj`. Never hand-edit the project.
- `setupStatusItem()` runs before any WebKit object is created (Sonoma NSStatusBar issue).
- API calls go through `callAsyncJavaScript`, never `evaluateJavaScript`, never `URLSession`.
- Signed in means a 2xx from `/api/organizations`, never cookie presence.
- Signed out signal: HTTP 401, or HTTP 403 whose body has `error.details.error_code == "account_session_invalid"`. A bare 403 is per-org.
- Only orgs whose `capabilities` contain `"chat"` may be selected.
- All test commands use the scratch DerivedData path so the build is reused: `DD=/private/tmp/claude-501/-Users-rahullalia-Developer-my-business-internal-tools-claude-usage-widget/2b2f1ff5-2095-46de-a2d4-92390a56b317/scratchpad/DerivedData` and `TESTCMD="xcodebuild test -scheme ClaudeUsageWidget -destination platform=macOS -derivedDataPath $DD CODE_SIGN_IDENTITY= CODE_SIGNING_REQUIRED=NO CODE_SIGNING_ALLOWED=NO"`.

- No em or en dashes in prose or docs; existing "—" placeholder glyphs inside UI string literals stay.

```swift
import Foundation

// MARK: - APIResponse

struct APIResponse: Equatable {
    let status: Int
    let body: Data

    var isSuccess: Bool { (200..<300).contains(status) }
}

// MARK: - ClaudeAPIError

enum ClaudeAPIError: Error, Equatable {
    case transport(String)     // webView gone, fetch threw, claude.ai did not load
    case sessionExpired        // 401, or 403 with error_code account_session_invalid
    case forbidden(String)     // any other 403, message from the envelope
    case notFound              // 404
    case server(Int)           // 5xx
    case http(Int)             // any other non-2xx
    case decode(String)        // body did not match the expected shape

    /// nil for 2xx. Pure.
    static func classify(_ response: APIResponse) -> ClaudeAPIError? {
        if response.isSuccess { return nil }
        let envelope = APIErrorEnvelope.parse(response.body)
        switch response.status {
        case 401:
            return .sessionExpired
        case 403:
            if envelope?.errorCode == "account_session_invalid" { return .sessionExpired }
            return .forbidden(envelope?.message ?? "Permission denied")
        case 404:
            return .notFound
        case 500...599:
            return .server(response.status)
        default:
            return .http(response.status)
        }
    }

    var userMessage: String {
        switch self {
        case .transport: return "Offline or claude.ai unreachable"
        case .sessionExpired: return "Session expired"
        case .forbidden: return "Permission denied for this org"
        case .notFound: return "Org no longer exists, sign out and back in"
        case .server(let code): return "claude.ai error \(code)"
        case .http(let code): return "claude.ai returned \(code)"
        case .decode: return "Unexpected response, please report"
        }
    }
}

// MARK: - Error envelope

/// {"type":"error","error":{"type":"...","message":"...","details":{"error_code":"..."}},"request_id":"..."}
struct APIErrorEnvelope: Decodable, Equatable {
    struct Inner: Decodable, Equatable {
        struct Details: Decodable, Equatable {
            let error_code: String?
        }
        let type: String
        let message: String?
        let details: Details?
    }

    let error: Inner

    var errorCode: String? { error.details?.error_code }
    var message: String? { error.message }

    static func parse(_ data: Data) -> APIErrorEnvelope? {
        try? JSONDecoder().decode(APIErrorEnvelope.self, from: data)
    }
}

// MARK: - ClaudeAPI

/// The only way the app talks to claude.ai. Implemented by ClaudeAPIClient (WKWebView) and by test fakes.
@MainActor
protocol ClaudeAPI: AnyObject {
    /// Same-origin GET. Throws only ClaudeAPIError.transport; HTTP errors come back as a response for classify().
    func get(_ path: String) async throws -> APIResponse
    /// Value of the lastActiveOrg cookie, if claude.ai has set one.
    func lastActiveOrgId() async -> String?
    /// Forget the loaded page; the next get() reloads claude.ai first. Call after sign-out.
    func reset()
}
```

```swift
import Foundation

/// One entry of GET /api/organizations. Only the fields the widget uses.
struct Organization: Decodable, Equatable {
    let uuid: String
    let name: String?
    let capabilities: [String]
    let rateLimitTier: String?
    let planDisplayName: String?

    private enum CodingKeys: String, CodingKey {
        case uuid, name, capabilities
        case rateLimitTier = "rate_limit_tier"
        case planDisplayName = "plan_display_name"
    }

    init(uuid: String, name: String?, capabilities: [String], rateLimitTier: String? = nil, planDisplayName: String? = nil) {
        self.uuid = uuid
        self.name = name
        self.capabilities = capabilities
        self.rateLimitTier = rateLimitTier
        self.planDisplayName = planDisplayName
    }

    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        uuid = try c.decode(String.self, forKey: .uuid)
        name = try c.decodeIfPresent(String.self, forKey: .name)
        capabilities = try c.decodeIfPresent([String].self, forKey: .capabilities) ?? []
        rateLimitTier = try c.decodeIfPresent(String.self, forKey: .rateLimitTier)
        planDisplayName = try c.decodeIfPresent(String.self, forKey: .planDisplayName)
    }

    /// Consumer (claude.ai chat) orgs carry "chat"; console orgs carry "api" and always 403 on usage.
    var isChatOrg: Bool { capabilities.contains("chat") }

    var displayName: String { name ?? "Claude" }

    /// The org claude.ai last had active, if it can chat; else the first chat org; else nil.
    static func pickChatOrg(from orgs: [Organization], lastActiveOrgId: String?) -> Organization? {
        let chatOrgs = orgs.filter(\.isChatOrg)
        if let id = lastActiveOrgId, let active = chatOrgs.first(where: { $0.uuid == id }) {
            return active
        }
        return chatOrgs.first
    }

    static func decodeList(from data: Data) throws -> [Organization] {
        try JSONDecoder().decode([Organization].self, from: data)
    }
}
```

        case .weeklyAll: return "Weekly (all models)"
        case .weeklyOpus: return "Weekly (Opus)"
        case .weeklySonnet: return "Weekly (Sonnet)"
        }
    }

    public var isWeekly: Bool { self != .session }

// MARK: - UsageBucket

public struct UsageBucket: Equatable {
    public let kind: BucketKind
    public let percentUsed: Double   // 0.0 to 1.0
    public let resetsAt: Date?
    public let lockedReason: String?

    public init(kind: BucketKind, percentUsed: Double, resetsAt: Date?, lockedReason: String?) {
        self.kind = kind
        self.percentUsed = percentUsed
        self.resetsAt = resetsAt
        self.lockedReason = lockedReason
    }

    /// "resets in 42m" or "resets in 1h 12m"; "–" if nil or past
    public var resetsInDisplay: String {
        guard let date = resetsAt else { return "–" }
        let remaining = date.timeIntervalSinceNow
        guard remaining > 0 else { return "–" }

        let totalMinutes = Int(remaining / 60)
        let hours = totalMinutes / 60
        let minutes = totalMinutes % 60

        if hours > 0 {
            return "resets in \(hours)h \(minutes)m"
        } else {
            return "resets in \(minutes)m"
        }
    }

    public var barColorState: RingColorState {
        .from(percent: percentUsed)
    }

    /// "Fri 9:00 AM"; nil if resetsAt is nil
    public var resetsAtDisplay: String? {
        guard let date = resetsAt else { return nil }
        let formatter = DateFormatter()
        formatter.dateFormat = "EEE h:mm a"
        return formatter.string(from: date)
    }
}

    public let percentUsed: Double?   // 0.0 to 1.0 when the API reports utilization

// MARK: - UsageData

public struct UsageData {
    /// Only the buckets the API returned with a utilization, in BucketKind order.
    public let buckets: [UsageBucket]
    public let extraUsage: ExtraUsage?
    public var lastUpdated: Date

    public init(buckets: [UsageBucket], extraUsage: ExtraUsage?, lastUpdated: Date = Date()) {
        self.buckets = buckets
        self.extraUsage = extraUsage
        self.lastUpdated = lastUpdated
    }

    public func bucket(_ kind: BucketKind) -> UsageBucket? {
        buckets.first { $0.kind == kind }
    }

    /// Ring value for a metric mode; 0 when the account has no such bucket.
    public func ringValue(for mode: RingMetricMode) -> Double {
        switch mode {
        case .session:
            return bucket(.session)?.percentUsed ?? 0.0
        case .weekly:
            return buckets.filter { $0.kind.isWeekly }.map(\.percentUsed).max() ?? 0.0
        }
    }

    public func ringColorState(for mode: RingMetricMode) -> RingColorState {
        .from(percent: ringValue(for: mode))
    }
}

// MARK: - JSON Decoding

private struct RawBucket: Decodable {
    let utilization: Double?
    let resets_at: Date?
    let locked_reason: String?
}

private struct RawExtraUsage: Decodable {
    let is_enabled: Bool?
    let utilization: Double?
}

/// Every key optional: plans differ in which buckets they return, and unknown keys are ignored.
private struct RawUsageResponse: Decodable {
    let five_hour: RawBucket?
    let seven_day: RawBucket?
    let seven_day_opus: RawBucket?
    let seven_day_sonnet: RawBucket?
    let extra_usage: RawExtraUsage?
}

extension UsageData {
    /// Decode from raw API JSON; handles fractional and whole-second ISO8601 dates.
    public static func decode(from data: Data) throws -> UsageData {
        let decoder = JSONDecoder()

        let fractionalFormatter = ISO8601DateFormatter()
        fractionalFormatter.formatOptions = [.withInternetDateTime, .withFractionalSeconds]

        let plainFormatter = ISO8601DateFormatter()
        plainFormatter.formatOptions = [.withInternetDateTime]

        decoder.dateDecodingStrategy = .custom { decoder in
            let container = try decoder.singleValueContainer()
            let string = try container.decode(String.self)
            if let date = fractionalFormatter.date(from: string) { return date }
            if let date = plainFormatter.date(from: string) { return date }
            throw DecodingError.dataCorruptedError(in: container, debugDescription: "Cannot decode date from: \(string)")
        }

        let raw = try decoder.decode(RawUsageResponse.self, from: data)

        let pairs: [(BucketKind, RawBucket?)] = [
            (.session, raw.five_hour),
            (.weeklyAll, raw.seven_day),
            (.weeklyOpus, raw.seven_day_opus),
            (.weeklySonnet, raw.seven_day_sonnet),
        ]
        var buckets: [UsageBucket] = []
        for (kind, rawBucket) in pairs {
            guard let rawBucket, let utilization = rawBucket.utilization else { continue }
            buckets.append(UsageBucket(
                kind: kind,
                percentUsed: utilization / 100.0,
                resetsAt: rawBucket.resets_at,
                lockedReason: rawBucket.locked_reason
            ))
        }

        let extra = raw.extra_usage.map {
            ExtraUsage(isEnabled: $0.is_enabled ?? false, percentUsed: $0.utilization.map { $0 / 100.0 })
        }

        return UsageData(buckets: buckets, extraUsage: extra, lastUpdated: Date())
    }
}

import Foundation
import WebKit

/// Talks to claude.ai through a hidden WKWebView so the session cookie rides along with every fetch.
/// The page is loaded once (claude.ai redirects to /login when signed out, same origin, fetches still work).
@MainActor
final class ClaudeAPIClient: NSObject, ClaudeAPI {

    private let webView: WKWebView
    private let baseURL = URL(string: "https://claude.ai")!
    private let loadTimeout: TimeInterval = 15

    private(set) var isReady = false
    private var isLoading = false
    private var readyContinuations: [CheckedContinuation<Void, Error>] = []
    private var timeoutTask: Task<Void, Never>?

    init(dataStore: WKWebsiteDataStore = .default()) {
        let config = WKWebViewConfiguration()
        config.websiteDataStore = dataStore
        webView = WKWebView(frame: .zero, configuration: config)
        super.init()
        webView.navigationDelegate = self
    }

    // MARK: - ClaudeAPI

    func get(_ path: String) async throws -> APIResponse {
        try await ensureReady()
        let js = """
        const response = await fetch(path, { credentials: 'include', signal: AbortSignal.timeout(15000) });
        const body = await response.text();
        return JSON.stringify({ status: response.status, body: body });
        """
        let result: Any?
        do {
            result = try await webView.callAsyncJavaScript(js, arguments: ["path": path], in: nil, contentWorld: .defaultClient)
        } catch {
            throw ClaudeAPIError.transport(error.localizedDescription)
        }
        return try Self.parseBridgeResult(result)
    }

    func lastActiveOrgId() async -> String? {
        let cookies = await webView.configuration.websiteDataStore.httpCookieStore.allCookies()
        return cookies.first { $0.name == "lastActiveOrg" && $0.domain.hasSuffix("claude.ai") }?.value
    }

    func reset() {
        isReady = false
    }

    // MARK: - Bridge result

    /// The JS returns JSON.stringify({status, body}); anything else is a transport failure.
    static func parseBridgeResult(_ result: Any?) throws -> APIResponse {
        guard let string = result as? String,
              let data = string.data(using: .utf8),
              let object = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let status = object["status"] as? Int,
              let body = object["body"] as? String else {
            throw ClaudeAPIError.transport("Bridge returned an unexpected value")
        }
        return APIResponse(status: status, body: Data(body.utf8))
    }

    // MARK: - Page readiness

    private func ensureReady() async throws {
        if isReady { return }
        if !isLoading {
            isLoading = true
            webView.load(URLRequest(url: baseURL))
            timeoutTask = Task { [weak self] in
                try? await Task.sleep(nanoseconds: UInt64(15 * 1_000_000_000))
                guard let self, !Task.isCancelled, !self.isReady else { return }
                self.finishLoading(.failure(ClaudeAPIError.transport("claude.ai did not load")))
            }
        }
        try await withCheckedThrowingContinuation { (continuation: CheckedContinuation<Void, Error>) in
            readyContinuations.append(continuation)
        }
    }

    private func finishLoading(_ result: Result<Void, Error>) {
        isLoading = false
        timeoutTask?.cancel()
        timeoutTask = nil
        if case .success = result { isReady = true }
        let waiters = readyContinuations
        readyContinuations.removeAll()
        waiters.forEach { $0.resume(with: result) }
    }

    func markReadyForTesting() {
        finishLoading(.success(()))
    }
}

// MARK: - WKNavigationDelegate

extension ClaudeAPIClient: WKNavigationDelegate {
    func webView(_ webView: WKWebView, didFinish navigation: WKNavigation!) {
        finishLoading(.success(()))
    }

    func webView(_ webView: WKWebView, didFail navigation: WKNavigation!, withError error: Error) {
        finishLoading(.failure(ClaudeAPIError.transport(error.localizedDescription)))
    }

    func webView(_ webView: WKWebView, didFailProvisionalNavigation navigation: WKNavigation!, withError error: Error) {
        finishLoading(.failure(ClaudeAPIError.transport(error.localizedDescription)))
    }
}

import AppKit
import WebKit

enum AuthState: Equatable {
    case unknown                                     // launch, first check in flight
    case unreachable                                 // claude.ai could not be reached or answered 5xx
    case signedOut                                   // no session (first launch, or after sign-out)
    case signedIn(Organization, chatOrgCount: Int)   // usage can be polled for this org
    case sessionExpired                              // was signed in, session no longer valid
    case noChatOrg                                   // valid session, but no org can chat (console-only account)
}

@MainActor
protocol AuthManagerDelegate: AnyObject {
    func authManager(_ manager: AuthManager, didChange state: AuthState)
}

/// Decides whether the app is signed in, by asking the API. Owns the login window.
@MainActor
final class AuthManager {
    weak var delegate: AuthManagerDelegate?

    private(set) var state: AuthState = .unknown {
        didSet { if oldValue != state { delegate?.authManager(self, didChange: state) } }
    }

    /// Tests set this to false; showLoginWindow() then only counts.
    var presentsLoginWindow = true
    private(set) var loginWindowRequests = 0

    private let client: ClaudeAPI
    private let dataStore: WKWebsiteDataStore
    private var loginWindow: LoginWindowController?
    private var expiryWindowShown = false

    init(client: ClaudeAPI, dataStore: WKWebsiteDataStore = .default()) {
        self.client = client
        self.dataStore = dataStore
    }

    // MARK: - Launch

    /// Launch path: find out who we are; with no session, open the login window.
    func checkAuthStatus() async {
        await verifySession()
        if state == .signedOut {
            showLoginWindow()
        }
    }

    /// GET /api/organizations and derive the state from the answer.
    @discardableResult
    func verifySession() async -> AuthState {
        do {
            let response = try await client.get("/api/organizations")
            if let error = ClaudeAPIError.classify(response) {
                apply(error)
                return state
            }
            let orgs = try Organization.decodeList(from: response.body)
            let activeId = await client.lastActiveOrgId()
            if let org = Organization.pickChatOrg(from: orgs, lastActiveOrgId: activeId) {
                expiryWindowShown = false
                state = .signedIn(org, chatOrgCount: orgs.filter(\.isChatOrg).count)
            } else {
                state = .noChatOrg
            }
        } catch let error as ClaudeAPIError {
            apply(error)
        } catch {
            apply(.decode(error.localizedDescription))
        }
        return state
    }

    private func apply(_ error: ClaudeAPIError) {
        switch error {
        case .sessionExpired:
            switch state {
            case .signedIn, .noChatOrg: state = .sessionExpired
            case .sessionExpired: break
            default: state = .signedOut
            }
        default:
            // Keep a signed-in (or signed-out) state through transient trouble; only the launch states move.
            if state == .unknown || state == .unreachable {
                state = .unreachable
            }
        }
    }

    // MARK: - Expiry

    /// Called by UsageService when a poll comes back session-invalid. Opens the login window once per expiry.
    func sessionDidExpire() {
        state = .sessionExpired
        guard !expiryWindowShown else { return }
        expiryWindowShown = true
        showLoginWindow()
    }

    // MARK: - Sign out

    func signOut() {
        dataStore.removeData(ofTypes: WKWebsiteDataStore.allWebsiteDataTypes(), modifiedSince: .distantPast) {
            Task { @MainActor [weak self] in
                guard let self else { return }
                self.client.reset()
                self.expiryWindowShown = false
                self.state = .signedOut
                self.showLoginWindow()
            }
        }
    }

    // MARK: - Login window

    func showLoginWindow() {
        loginWindowRequests += 1
        guard presentsLoginWindow else { return }
        if loginWindow == nil {
            loginWindow = LoginWindowController(authManager: self)
        }
        loginWindow?.showWindow(nil)
        loginWindow?.window?.makeKeyAndOrderFront(nil)
        NSApp.activate(ignoringOtherApps: true)
    }

    /// The login window saw a URL that may mean success, or its fallback timer fired. True once the API agrees.
    func loginWindowSawCandidate() async -> Bool {
        switch await verifySession() {
        case .signedIn, .noChatOrg: return true
        default: return false
        }
    }

    func loginWindowDidClose() {
        loginWindow = nil
    }
}
```

import AppKit
import WebKit

class LoginWindowController: NSWindowController, WKNavigationDelegate, WKUIDelegate, NSWindowDelegate {
    weak var authManager: AuthManager?
    private var webView: WKWebView!
    private var verifying = false
    private var finished = false
    private var fallbackTimer: Timer?

    // Holds any OAuth popup windows (e.g. Google sign-in)
    private var popupWindowControllers: [NSWindowController] = []

    init(authManager: AuthManager?) {
        let window = NSWindow(
            contentRect: NSRect(x: 0, y: 0, width: 520, height: 640),
            styleMask: [.titled, .closable, .resizable],
            backing: .buffered,
            defer: false
        )
        window.title = "Sign in to Claude"
        window.center()
        super.init(window: window)
        self.authManager = authManager
        window.delegate = self
        setupWebView()
    }

    required init?(coder: NSCoder) { fatalError("init(coder:) not implemented") }

    deinit {
        webView?.removeObserver(self, forKeyPath: "URL")
    }

    private func setupWebView() {
        let config = WKWebViewConfiguration()
        config.websiteDataStore = WKWebsiteDataStore.default()
        config.preferences.javaScriptCanOpenWindowsAutomatically = true
        webView = WKWebView(frame: .zero, configuration: config)
        webView.navigationDelegate = self
        webView.uiDelegate = self
        webView.addObserver(self, forKeyPath: "URL", options: .new, context: nil)
        window?.contentView = webView
        webView.load(URLRequest(url: URL(string: "https://claude.ai/login")!))
    }

    override func showWindow(_ sender: Any?) {
        super.showWindow(sender)
        finished = false
        fallbackTimer?.invalidate()
        // Fallback: a login that never changes the URL still gets detected within 2 s.
        fallbackTimer = Timer.scheduledTimer(withTimeInterval: 2, repeats: true) { [weak self] _ in
            self?.verify()
        }
    }

    // MARK: - Detection: URL changes only trigger an API check; the API decides.

    override func observeValue(
        forKeyPath keyPath: String?,
        of object: Any?,
        change: [NSKeyValueChangeKey: Any]?,
        context: UnsafeMutableRawPointer?
    ) {
        if keyPath == "URL", let url = webView?.url {
            noteCandidate(url: url)
        }
    }

    func webView(_ webView: WKWebView, didFinish navigation: WKNavigation!) {
        if let url = webView.url { noteCandidate(url: url) }
        if webView !== self.webView, let mainURL = self.webView?.url { noteCandidate(url: mainURL) }
    }

    /// A claude.ai URL outside the login routes. Not proof of a session, only a reason to ask the API.
    func isCandidateLoginURL(_ url: URL) -> Bool {
        guard let host = url.host, host.contains("claude.ai") else { return false }
        let loginPaths = ["/login", "/auth", "/oauth"]
        return !loginPaths.contains(where: { url.path.hasPrefix($0) })
    }

    private func noteCandidate(url: URL) {
        guard isCandidateLoginURL(url) else { return }
        verify()
    }

    private func verify() {
        guard !verifying, !finished, let authManager else { return }
        verifying = true
        Task { @MainActor [weak self] in
            let confirmed = await authManager.loginWindowSawCandidate()
            guard let self else { return }
            self.verifying = false
            if confirmed { self.finish() }
        }
    }

    private func finish() {
        finished = true
        fallbackTimer?.invalidate()
        fallbackTimer = nil
        popupWindowControllers.forEach { $0.close() }
        popupWindowControllers.removeAll()
        close()
    }

    // MARK: - NSWindowDelegate

    func windowWillClose(_ notification: Notification) {
        fallbackTimer?.invalidate()
        fallbackTimer = nil
        authManager?.loginWindowDidClose()
    }

import Foundation

@MainActor
protocol UsageServiceDelegate: AnyObject {
    func usageService(_ service: UsageService, didUpdate data: UsageData)
    func usageService(_ service: UsageService, didFailWith error: ClaudeAPIError)
    func usageServiceDidDetectSessionExpiry(_ service: UsageService)
}

/// Polls one org's usage every five minutes. Stops itself and reports when the session is gone.
@MainActor
final class UsageService {
    weak var delegate: UsageServiceDelegate?

    let orgId: String
    var pollInterval: TimeInterval = 300
    private(set) var lastData: UsageData?
    private(set) var isPolling = false

    private let client: ClaudeAPI
    private var timer: Timer?

    init(client: ClaudeAPI, orgId: String) {
        self.client = client
        self.orgId = orgId
    }

    func start() {
        stop()
        isPolling = true
        fetch()
        timer = Timer.scheduledTimer(withTimeInterval: pollInterval, repeats: true) { [weak self] _ in
            Task { @MainActor in self?.fetch() }
        }
    }

    func stop() {
        timer?.invalidate()
        timer = nil
        isPolling = false
    }

    /// Manual refresh trigger
    func fetch() {
        Task { await fetchNow() }
    }

    func fetchNow() async {
        do {
            let response = try await client.get("/api/organizations/\(orgId)/usage")
            if let error = ClaudeAPIError.classify(response) {
                if error == .sessionExpired {
                    stop()
                    delegate?.usageServiceDidDetectSessionExpiry(self)
                } else {
                    delegate?.usageService(self, didFailWith: error)
                }
                return
            }
            let data = try UsageData.decode(from: response.body)
            lastData = data
            delegate?.usageService(self, didUpdate: data)
        } catch let error as ClaudeAPIError {
            delegate?.usageService(self, didFailWith: error)
        } catch {
            delegate?.usageService(self, didFailWith: .decode(error.localizedDescription))
        }
    }
}
```
